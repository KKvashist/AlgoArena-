import { BrowserRouter, Routes, Route, Navigate } from 'react-router-dom'
import Navbar from './components/Navbar'
import BackgroundFX from './components/BackgroundFX'
import StatusBar from './components/StatusBar'
import ProtectedRoute from './components/ProtectedRoute'
import ArenaDock from './components/ArenaDock'
import { ArenaProvider, ARENA_WIDTHS, useArena } from './context/ArenaContext'
import { useAuth } from './context/AuthContext'
import Home from './pages/Home'
import Login from './pages/Login'
import Signup from './pages/Signup'
import Dashboard from './pages/Dashboard'
import Notes from './pages/Notes'
import TopicPage from './pages/TopicPage'

// Login / Signup pages: agar user already logged in hai to seedha Home bhej do
function PublicOnly({ children }: { children: ReactElement }) {
  const { user } = useAuth()
  if (user) return <Navigate to="/" replace />
  return children
}

function Shell() {
  const { user } = useAuth()
  const { open, size } = useArena()

  // Full screen covers the page, so no shift is needed in that mode
  const shift = user && open && size !== 'full' ? ARENA_WIDTHS[size] : '0px'

  return (
    <>
      <BackgroundFX />

      {/* The whole page shifts left when the panel opens (large screens) */}
      <div
        className="relative z-10 transition-[margin] duration-300 ease-out lg:mr-[var(--arena-w)]"
        style={{ '--arena-w': shift } as CSSProperties}
      >
        <Navbar />
        <StatusBar />
        <Routes>
          {/* Public: sirf login / signup */}
          <Route
            path="/login"
            element={
              <PublicOnly>
                <Login />
              </PublicOnly>
            }
          />
          <Route
            path="/signup"
            element={
              <PublicOnly>
                <Signup />
              </PublicOnly>
            }
          />

          {/* Baaki sab kuch login ke peeche. "/" bhi, taaki site khulte hi login aaye */}
          <Route element={<ProtectedRoute />}>
            <Route path="/" element={<Home />} />
            <Route path="/dashboard" element={<Dashboard />} />
            <Route path="/notes" element={<Notes />} />
            <Route path="/topic/:slug" element={<TopicPage />} />
          </Route>

          {/* Unknown URL -> Home (jo login na ho to login par jayega) */}
          <Route path="*" element={<Navigate to="/" replace />} />
        </Routes>
      </div>

      {user && <ArenaDock key={user.id} />}
    </>
  )
}

function App() {
  return (
    <BrowserRouter>
      <ArenaProvider>
        <Shell />
      </ArenaProvider>
    </BrowserRouter>
  )
}

export default App

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useState,
  type ReactNode,
} from 'react'

export type ArenaSize = 'normal' | 'wide' | 'full'

// Panel widths. The page shifts by exactly this amount when the panel opens.
// eslint-disable-next-line react-refresh/only-export-components
export const ARENA_WIDTHS: Record<ArenaSize, string> = {
  normal: 'min(440px, 100vw)',
  wide: 'min(900px, 60vw)',
  full: '100vw',
}

type ArenaCtx = {
  open: boolean
  size: ArenaSize
  setSize: (s: ArenaSize) => void
  openArena: () => void
  closeArena: () => void
  toggleArena: () => void
}

const ArenaContext = createContext<ArenaCtx | null>(null)

export function ArenaProvider({ children }: { children: ReactNode }) {
  const [open, setOpen] = useState(false)
  const [size, setSize] = useState<ArenaSize>('normal')

  const openArena = useCallback(() => setOpen(true), [])
  const closeArena = useCallback(() => setOpen(false), [])
  const toggleArena = useCallback(() => setOpen((v) => !v), [])

  useEffect(() => {
    function onKey(e: KeyboardEvent) {
      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') {
        e.preventDefault()
        setOpen((v) => !v)
      } else if (e.key === 'Escape') {
        setOpen(false)
      }
    }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [])

  return (
    <ArenaContext.Provider
      value={{ open, size, setSize, openArena, closeArena, toggleArena }}
    >
      {children}
    </ArenaContext.Provider>
  )
}

// eslint-disable-next-line react-refresh/only-export-components
export function useArena() {
  const ctx = useContext(ArenaContext)
  if (!ctx) throw new Error('useArena must be used inside ArenaProvider')
  return ctx
}

  email: string
  xp: number

type AuthCtx = {
  user: User | null
  loading: boolean
  login: (email: string, password: string) => Promise<void>
  signup: (name: string, email: string, password: string) => Promise<void>
  logout: () => void
}

const AuthContext = createContext<AuthCtx | null>(null)
const TOKEN_KEY = 'algoarena_token'

async function request(url: string, body: object) {
  const res = await fetch(url, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  })
  const data = await res.json().catch(() => ({}))
  if (!res.ok) throw new Error(data.error || 'Something went wrong')
  return data as { token: string; user: User }
}

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<User | null>(null)
  const [loading, setLoading] = useState(true)

  // Page refresh ke baad token se user wapas laao
  useEffect(() => {
    const token = localStorage.getItem(TOKEN_KEY)
    if (!token) {
      setLoading(false)
      return
    }
    fetch('/api/auth/me', { headers: { Authorization: `Bearer ${token}` } })
      .then(async (res) => {
        if (!res.ok) throw new Error()
        const data = await res.json()
        setUser(data.user)
      })
      .catch(() => localStorage.removeItem(TOKEN_KEY))
      .finally(() => setLoading(false))
  }, [])

  async function login(email: string, password: string) {
    const data = await request('/api/auth/login', { email, password })
    localStorage.setItem(TOKEN_KEY, data.token)
    setUser(data.user)
  }

  async function signup(name: string, email: string, password: string) {
    const data = await request('/api/auth/signup', { name, email, password })
    localStorage.setItem(TOKEN_KEY, data.token)
    setUser(data.user)
  }

  function logout() {
    localStorage.removeItem(TOKEN_KEY)
    setUser(null)
  }

  return (
    <AuthContext.Provider value={{ user, loading, login, signup, logout }}>
      {children}
    </AuthContext.Provider>
  )
}

// eslint-disable-next-line react-refresh/only-export-components
export function useAuth() {
  const ctx = useContext(AuthContext)
  if (!ctx) throw new Error('useAuth must be used inside AuthProvider')
  return ctx
}

function prefixCells(arr: number[]): FillCell[] {
  const out: FillCell[] = []
  let run = 0
  arr.forEach((v, i) => {
    const prev = run
    run += v
    out.push({
      label: `prefix[${i}]`,
      top: `arr[${i}] = ${v}`,
      given: i === 0 ? run : undefined,
      answer: run,
      rule: i === 0 ? 'prefix[0] = arr[0]' : `prefix[${i}] = prefix[${i - 1}] + arr[${i}]`,
      work: i === 0 ? `${v}` : `${prev} + ${v} = ${run}`,
    })
  })
  return out
}

function coinCells(coins: number[], upto: number): FillCell[] {
  const dp: number[] = [0]
  const out: FillCell[] = [{ label: 'dp[0]', top: 'amount 0', given: 0, answer: 0, rule: 'base case', work: '0 coins' }]
  for (let i = 1; i <= upto; i++) {
    const usable = coins.filter((c) => c <= i)
    const opts = usable.map((c) => dp[i - c])
    const best = Math.min(...opts) + 1
    dp[i] = best
    out.push({
      label: `dp[${i}]`,
      top: `amount ${i}`,
      given: i <= 2 ? best : undefined,
      answer: best,
      rule: `dp[${i}] = 1 + min(${usable.map((c) => `dp[${i - c}]`).join(', ')})`,
      work: `1 + min(${opts.join(', ')}) = ${best}`,
    })
  }
  return out
}

function fibCells(n: number): FillCell[] {
  const dp: number[] = [0, 1]
  const out: FillCell[] = [
    { label: 'dp[0]', given: 0, answer: 0, rule: 'base case', work: '0' },
    { label: 'dp[1]', given: 1, answer: 1, rule: 'base case', work: '1' },
  ]
  for (let i = 2; i <= n; i++) {
    dp[i] = dp[i - 1] + dp[i - 2]
    out.push({ label: `dp[${i}]`, answer: dp[i], rule: `dp[${i}] = dp[${i - 1}] + dp[${i - 2}]`, work: `${dp[i - 1]} + ${dp[i - 2]} = ${dp[i]}` })
  }
  return out
}

import { useEffect, useState } from 'react'

function useCountUp(target: number, duration = 1000) {
  const [value, setValue] = useState(0)

  useEffect(() => {
    let startTime: number | null = null

    function step(timestamp: number) {
      if (startTime === null) startTime = timestamp
      const progress = Math.min((timestamp - startTime) / duration, 1)
      setValue(Math.floor(progress * target))
      if (progress < 1) requestAnimationFrame(step)
    }

    const frame = requestAnimationFrame(step)
    return () => cancelAnimationFrame(frame)
  }, [target, duration])

  return value
}

export default useCountUp

import { useEffect, useState } from 'react'

function useTypewriter(phrases: string[], typingSpeed = 60, pauseTime = 1800) {
  const [text, setText] = useState('')
  const [phraseIndex, setPhraseIndex] = useState(0)
  const [isDeleting, setIsDeleting] = useState(false)

  useEffect(() => {
    const currentPhrase = phrases[phraseIndex]
    let timeout: ReturnType<typeof setTimeout>

    if (!isDeleting && text === currentPhrase) {
      timeout = setTimeout(() => setIsDeleting(true), pauseTime)
    } else if (isDeleting && text === '') {
      timeout = setTimeout(() => {
        setIsDeleting(false)
        setPhraseIndex((prev) => (prev + 1) % phrases.length)
      }, typingSpeed)
    } else {
      const nextText = isDeleting
        ? currentPhrase.slice(0, text.length - 1)
        : currentPhrase.slice(0, text.length + 1)

      timeout = setTimeout(
        () => setText(nextText),
        isDeleting ? typingSpeed / 2 : typingSpeed
      )
    }

    return () => clearTimeout(timeout)
  }, [text, isDeleting, phraseIndex, phrases, typingSpeed, pauseTime])

  return text
}

export default useTypewriter

  useEffect(() => {
    try { localStorage.setItem(key, JSON.stringify(solved)); } catch { /* ignore */ }
  }, [key, solved]);

  const toggle = (name: string) => setSolved((s) => (s.includes(name) ? s.filter((x) => x !== name) : [...s, name]));

  return (
    <div className="tp-card">
      <p className="tp-muted">{solved.length}/{topic.problems.length} solved</p>
      <ul className="tp-problems">
        {topic.problems.map((p) => (
          <li key={p.name}>
            <label>
              <input type="checkbox" checked={solved.includes(p.name)} onChange={() => toggle(p.name)} />
              <a href={p.url} target="_blank" rel="noreferrer">{p.name}</a>
            </label>
            <span className={`tp-diff ${p.difficulty.toLowerCase()}`}>{p.difficulty}</span>
          </li>
        ))}
      </ul>
    </div>
  );
}

function QuizGame({ topic }: { topic: Topic }) {
  const questions = useMemo(() => topic.quiz, [topic]);
  const [i, setI] = useState(0);
  const [picked, setPicked] = useState<number | null>(null);
  const [score, setScore] = useState(0);
  const [done, setDone] = useState(false);

  const q = questions[i];

  const choose = (idx: number) => {
    if (picked !== null) return;
    setPicked(idx);
    if (idx === q.answer) setScore((s) => s + 1);
  };

  const next = () => {
    if (i + 1 >= questions.length) return setDone(true);
    setI(i + 1);
    setPicked(null);
  };

  const restart = () => { setI(0); setPicked(null); setScore(0); setDone(false); };

  if (done) {
    return (
      <div className="tp-card tp-center">
        <h3>Round complete</h3>
        <p className="tp-score">{score}/{questions.length}</p>
        <p className="tp-muted">+{score * 10} XP earned</p>
        <button className="tp-btn" onClick={restart}>Play again</button>
      </div>
    );
  }

  return (
    <div className="tp-card">
      <p className="tp-muted">Question {i + 1} of {questions.length} · Score {score}</p>
      <h3>{q.q}</h3>
      <div className="tp-options">
        {q.options.map((o, idx) => {
          let cls = "";
          if (picked !== null) {
            if (idx === q.answer) cls = "correct";
            else if (idx === picked) cls = "wrong";
          }
          return <button key={o} className={cls} onClick={() => choose(idx)}>{o}</button>;
        })}
      </div>
      {picked !== null && (
        <>
          <p className="tp-why">{picked === q.answer ? "✅ Correct. " : "❌ Not quite. "}{q.why}</p>
          <button className="tp-btn" onClick={next}>{i + 1 >= questions.length ? "Finish" : "Next →"}</button>
        </>
      )}
    </div>
  );
}

// Middleware for protected routes
export function requireAuth(req, res, next) {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : null;
  if (!token) return res.status(401).json({ error: "Login required." });
  try {
    req.userId = jwt.verify(token, getSecret()).id;
    next();
  } catch {
    res.status(401).json({ error: "Session expired. Please log in again." });
  }
}

  if (name.length < 2) return res.status(400).json({ error: "Name must be at least 2 characters." });
  if (!emailOk(email)) return res.status(400).json({ error: "Please enter a valid email address." });
  if (password.length < 6) return res.status(400).json({ error: "Password must be at least 6 characters." });

  const users = readUsers();
  if (users.some((u) => u.email === email)) {
    return res.status(409).json({ error: "An account with this email already exists. Please log in." });
  }

  const user = {
    id: Date.now().toString(36) + Math.random().toString(36).slice(2, 7),
    name,
    email,
    passwordHash: await bcrypt.hash(password, 10),
    xp: 0,
    level: 1,
    createdAt: new Date().toISOString(),
  };

  writeUsers(users);

  res.status(201).json({ token: signToken(user), user: publicUser(user) });
});

// Retries automatically when Gemini is busy (503) or rate limited (429)
async function generateWithRetry(params, tries = 4) {
  for (let i = 0; i < tries; i++) {
    try {
      return await ai.models.generateContent(params);
    } catch (err) {
      const retryable = err?.status === 503 || err?.status === 429;
      if (!retryable || i === tries - 1) throw err;
      await sleep(1500 * (i + 1));
    }
  }
}

}

  let sent = false;

  for (const model of models) {
      for (let i = 0; i < 3; i++) {
      try {
        const stream = await ai.models.generateContentStream({
          model,
          contents,
          config: {
            systemInstruction: SYSTEM_PROMPT,
            temperature: 0.6,
            maxOutputTokens: 2048,
          },
        });

        for await (const chunk of stream) {
          if (!chunk.text) continue;
          if (!sent) {
            res.setHeader("Content-Type", "text/plain; charset=utf-8");
            res.setHeader("Cache-Control", "no-cache");
            sent = true;
          }
          res.write(chunk.text);
        }

        if (!sent) {
          return res.status(502).json({ error: EMPTY_MSG });
        }
        return res.end();
      } catch (err) {
        console.error("STREAM ERROR:", model, err?.status, err?.message);
        if (sent) return res.end(); // Connection broke midway: keep what already arrived
        const retryable = err?.status === 503 || err?.status === 429;
        if (err?.status === 404) break; // Model unavailable: try the next one
        if (!retryable) {
          return res.status(500).json({ error: ERROR_MSG });
        }
                await sleep(2000);
      }
    }
  }

  if (!sent) {
    res.status(503).json({ error: BUSY_MSG });
  }
